           ];

        // Fallback placeholder image handler
        function handleImageError(img, index) {
            img.onerror = null;
            img.src = `https://placehold.co/600x400/fef3c7/d97706?text=12A4+K%E1%BB%B7+Ni%E1%BB%87m+${index + 1}`;
        }

        // --- 1. RENDER & QUẢN LÝ 3D CAROUSEL GALLERY ---
        const carouselEl = document.getElementById('carousel');
        const indicatorEl = document.getElementById('carousel-indicator');
        const captionDisplayEl = document.getElementById('current-caption-display');
        let currentIndex = 0;
        const totalCards = galleryData.length;

        function initCarousel() {
            carouselEl.innerHTML = '';
            galleryData.forEach((item, index) => {
                const card = document.createElement('div');
                card.className = 'carousel-card';
                card.innerHTML = `
                    <img src="${item.src}" onerror="handleImageError(this, ${index})" class="w-full h-full object-cover" alt="${item.caption}">
                    <div class="absolute inset-x-0 bottom-0 bg-gradient-to-t from-black/85 via-black/40 to-transparent p-3 text-white text-xs font-semibold text-center">
                        ${item.caption}
                    </div>
                `;
                card.addEventListener('click', () => {
                    if (index === currentIndex) {
                        openImageModal(item.src, item.caption, index);
                    } else {
                        currentIndex = index;
                        updateCarousel();
                    }
                });
                carouselEl.appendChild(card);
            });
            updateCarousel();
        }

        function updateCarousel() {
            const cards = carouselEl.children;
            const angleStep = 360 / totalCards;
            const radius = window.innerWidth < 640 ? 250 : 360; // Bán kính quay 3D cho 10 thẻ

            for (let i = 0; i < cards.length; i++) {
                const offset = i - currentIndex;
                const angle = offset * angleStep;
                
                // Tính toán vị trí transform 3D xoay vòng tròn
                cards[i].style.transform = `rotateY(${angle}deg) translateZ(${radius}px)`;
                
                if (i === currentIndex) {
                    cards[i].style.opacity = '1';
                    cards[i].style.zIndex = '25';
                    cards[i].style.filter = 'none';
                    cards[i].style.border = '4px solid #f59e0b';
                } else {
                    cards[i].style.opacity = Math.abs(offset) > 3 ? '0.2' : '0.65';
                    cards[i].style.zIndex = '5';
                    cards[i].style.filter = 'brightness(0.7)';
                    cards[i].style.border = '3px solid white';
                }
            }
            indicatorEl.innerText = `${currentIndex + 1} / ${totalCards}`;
            captionDisplayEl.innerText = galleryData[currentIndex].caption;
        }

        document.getElementById('btn-prev').addEventListener('click', () => {
            currentIndex = (currentIndex - 1 + totalCards) % totalCards;
            updateCarousel();
        });

        document.getElementById('btn-next').addEventListener('click', () => {
            currentIndex = (currentIndex + 1) % totalCards;
            updateCarousel();
        });

        function openImageModal(src, caption, index) {
            modalImg.src = src;
            modalImg.onerror = function() {
                handleImageError(this, index);
            };
            modalCaption.innerText = caption;
            modal.classList.remove('hidden');
        }

        document.getElementById('btn-close-modal').addEventListener('click', () => {
            modal.classList.add('hidden');
        });
        modal.addEventListener('click', (e) => {
            if (e.target === modal) modal.classList.add('hidden');
        });

                document.body.appendChild(sparkle);

            }
        });
